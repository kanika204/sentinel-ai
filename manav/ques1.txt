import pandas as pd
import matplotlib.pyplot as plt

# 1. Create sample data
data = {
    "Timestamp": [
        "10:00", "10:01", "10:02", "10:03", "10:04",
        "10:05", "10:06", "10:07", "10:08", "10:09",
        "10:10", "10:11", "10:12", "10:13", "10:14",
        "10:15", "10:16", "10:17", "10:18", "10:19"
    ],

    "CPU": [
        45, 50, 52, 48, 55,
        95, 60, 58, 54, 57,
        62, 65, 97, 61, 59,
        55, 60, 58, 92, 56
    ],

    "Memory": [
        60, 62, 61, 63, 64,
        70, 65, 66, 64, 63,
        68, 70, 75, 67, 65,
        64, 63, 62, 70, 65
    ],

    "Response_Time": [
        120, 130, 125, 140, 135,
        500, 150, 145, 140, 130,
        135, 145, 550, 140, 135,
        130, 125, 140, 450, 130
    ]
}

# 2. Create DataFrame
df = pd.DataFrame(data)

print("Total records:", len(df))

# 3. Basic statistics
print("\nStatistics:")
print(df[["CPU", "Memory", "Response_Time"]].describe())

# 4. Detect anomalies
anomalies = df[
    (df["CPU"] > 80) |
    (df["Memory"] > 90) |
    (df["Response_Time"] > 400)
]

print("\nAnomalies:")
print(anomalies)

print("\nAnomalies detected:", len(anomalies))

# 5. Plot CPU
plt.plot(df["Timestamp"], df["CPU"], marker="o")

# Highlight anomalies
plt.scatter(
    anomalies["Timestamp"],
    anomalies["CPU"],
    color="red"
)

plt.axhline(80, linestyle="--")

plt.xlabel("Time")
plt.ylabel("CPU Usage")
plt.title("CPU Usage and Anomalies")
plt.xticks(rotation=45)
plt.show()
